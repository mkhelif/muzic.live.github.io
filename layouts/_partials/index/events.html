{{- /* Cached index of events, grouped by the entities they reference. Returns a scratch with three maps: - "artists" : artist id -> slice of event pages - "venues" : venue id -> slice of event pages - "festivals" : festival id -> slice of event pages Each slice is sorted by date (ascending), like the previous `.ByDate`. */ -}} {{- $cache := newScratch -}} {{- $cache.Set "artists" dict -}} {{- $cache.Set "venues" dict -}} {{- $cache.Set "festivals" dict -}} {{- range $event := (where site.RegularPages "Section" "events").ByDate -}} {{- range $event.Params.artists -}} {{- $existing := index ($cache.Get "artists") . | default slice -}} {{- $cache.SetInMap "artists" . ($existing | append $event) -}} {{- end -}} {{- with $event.Params.venue -}} {{- $existing := index ($cache.Get "venues") . | default slice -}} {{- $cache.SetInMap "venues" . ($existing | append $event) -}} {{- end -}} {{- with $event.Params.festival -}} {{- $existing := index ($cache.Get "festivals") . | default slice -}} {{- $cache.SetInMap "festivals" . ($existing | append $event) -}} {{- end -}} {{- end -}} {{- return $cache -}}
