{{- /* Cached index of events, grouped by the entities they reference. Returns a scratch with three maps: - "artists" : artist id -> slice of event pages - "venues" : venue id -> slice of event pages - "festivals" : festival id -> slice of event pages Each slice is sorted by date (ascending), like the previous `.ByDate`. */ -}} {{- $cache := newScratch -}} {{- $byArtist := dict -}} {{- $byVenue := dict -}} {{- $byFestival := dict -}} {{- range (where site.RegularPages "Section" "events").ByDate -}} {{- $event := . -}} {{- range .Params.artists -}} {{- $id := . -}} {{- $current := index $byArtist $id | default slice -}} {{- $byArtist = merge $byArtist (dict $id ($current | append $event)) -}} {{- end -}} {{- with .Params.venue -}} {{- $id := . -}} {{- $current := index $byVenue $id | default slice -}} {{- $byVenue = merge $byVenue (dict $id ($current | append $event)) -}} {{- end -}} {{- with .Params.festival -}} {{- $id := . -}} {{- $current := index $byFestival $id | default slice -}} {{- $byFestival = merge $byFestival (dict $id ($current | append $event)) -}} {{- end -}} {{- end -}} {{- $cache.Set "artists" $byArtist -}} {{- $cache.Set "venues" $byVenue -}} {{- $cache.Set "festivals" $byFestival -}} {{- return $cache -}}
