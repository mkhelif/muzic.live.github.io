{{- /* Cached index of reports (chroniques), grouped by referenced entity. Returns a scratch with: - "artists" : artist id -> slice of report pages - "venues" : venue id -> slice of report pages - "festivals" : festival id -> slice of report pages Slices are sorted by date, most recent first (previous `.ByDate.Reverse`). */ -}} {{- $cache := newScratch -}} {{- $cache.Set "artists" dict -}} {{- $cache.Set "venues" dict -}} {{- $cache.Set "festivals" dict -}} {{- range $report := (where site.RegularPages "Section" "reports").ByDate.Reverse -}} {{- range $report.Params.artists -}} {{- $existing := index ($cache.Get "artists") . | default slice -}} {{- $cache.SetInMap "artists" . ($existing | append $report) -}} {{- end -}} {{- with $report.Params.venue -}} {{- $existing := index ($cache.Get "venues") . | default slice -}} {{- $cache.SetInMap "venues" . ($existing | append $report) -}} {{- end -}} {{- with $report.Params.festival -}} {{- $existing := index ($cache.Get "festivals") . | default slice -}} {{- $cache.SetInMap "festivals" . ($existing | append $report) -}} {{- end -}} {{- end -}} {{- return $cache -}}
