{{- /* Cached index of reports (chroniques), grouped by referenced entity. Returns a scratch with: - "artists" : artist id -> slice of report pages - "venues" : venue id -> slice of report pages - "festivals" : festival id -> slice of report pages Slices are sorted by date, most recent first (previous `.ByDate.Reverse`). */ -}} {{- $cache := newScratch -}} {{- $byArtist := dict -}} {{- $byVenue := dict -}} {{- $byFestival := dict -}} {{- range (where site.RegularPages "Section" "reports").ByDate.Reverse -}} {{- $report := . -}} {{- range .Params.artists -}} {{- $id := . -}} {{- $current := index $byArtist $id | default slice -}} {{- $byArtist = merge $byArtist (dict $id ($current | append $report)) -}} {{- end -}} {{- with .Params.venue -}} {{- $id := . -}} {{- $current := index $byVenue $id | default slice -}} {{- $byVenue = merge $byVenue (dict $id ($current | append $report)) -}} {{- end -}} {{- with .Params.festival -}} {{- $id := . -}} {{- $current := index $byFestival $id | default slice -}} {{- $byFestival = merge $byFestival (dict $id ($current | append $report)) -}} {{- end -}} {{- end -}} {{- $cache.Set "artists" $byArtist -}} {{- $cache.Set "venues" $byVenue -}} {{- $cache.Set "festivals" $byFestival -}} {{- return $cache -}}
