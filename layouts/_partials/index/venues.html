{{- /* Cached parent -> direct children index for venues. Returns a scratch whose only map "children" maps a parent venue id to the slice of its direct children ids. */ -}} {{- $cache := newScratch -}} {{- $cache.Set "children" dict -}} {{- range where site.Pages "Section" "venues" -}} {{- $id := .Params.id -}} {{- with .Params.venue -}} {{- if $id -}} {{- $existing := index ($cache.Get "children") . | default slice -}} {{- $cache.SetInMap "children" . ($existing | append $id) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $cache -}}
