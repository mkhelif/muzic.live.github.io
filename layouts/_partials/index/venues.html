{{- /* Cached parent -> direct children index for venues. Returns a scratch whose only map "children" maps a parent venue id to the slice of its direct children ids. */ -}} {{- $cache := newScratch -}} {{- $children := dict -}} {{- range where site.Pages "Section" "venues" -}} {{- $id := .Params.id -}} {{- $parent := .Params.venue -}} {{- if and $id $parent -}} {{- $current := index $children $parent | default slice -}} {{- $children = merge $children (dict $parent ($current | append $id)) -}} {{- end -}} {{- end -}} {{- $cache.Set "children" $children -}} {{- return $cache -}}
