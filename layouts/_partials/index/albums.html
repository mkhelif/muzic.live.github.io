{{- /* Cached index of album reviews, grouped by artist id. Returns a scratch with: - "artists" : artist id -> slice of album pages (most recent first) */ -}} {{- $cache := newScratch -}} {{- $cache.Set "artists" dict -}} {{- range $album := (where site.RegularPages "Section" "albums").ByDate.Reverse -}} {{- range $album.Params.artists -}} {{- $existing := index ($cache.Get "artists") . | default slice -}} {{- $cache.SetInMap "artists" . ($existing | append $album) -}} {{- end -}} {{- end -}} {{- return $cache -}}
