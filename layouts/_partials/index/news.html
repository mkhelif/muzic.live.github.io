{{- /* Cached index of news, grouped by referenced entity. Returns a scratch with: - "artists" : artist id -> slice of news pages - "festivals" : festival id -> slice of news pages - "venues" : venue id -> slice of news pages - "albums" : album id -> slice of news pages Slices are sorted by date, most recent first. */ -}} {{- $cache := newScratch -}} {{- $byArtist := dict -}} {{- $byFestival := dict -}} {{- $byVenue := dict -}} {{- $byAlbum := dict -}} {{- range (where site.RegularPages "Section" "news").ByDate.Reverse -}} {{- $news := . -}} {{- range .Params.artists -}} {{- $id := . -}} {{- $current := index $byArtist $id | default slice -}} {{- $byArtist = merge $byArtist (dict $id ($current | append $news)) -}} {{- end -}} {{- with .Params.festival -}} {{- $id := . -}} {{- $current := index $byFestival $id | default slice -}} {{- $byFestival = merge $byFestival (dict $id ($current | append $news)) -}} {{- end -}} {{- with .Params.venue -}} {{- $id := . -}} {{- $current := index $byVenue $id | default slice -}} {{- $byVenue = merge $byVenue (dict $id ($current | append $news)) -}} {{- end -}} {{- with .Params.album -}} {{- $id := . -}} {{- $current := index $byAlbum $id | default slice -}} {{- $byAlbum = merge $byAlbum (dict $id ($current | append $news)) -}} {{- end -}} {{- end -}} {{- $cache.Set "artists" $byArtist -}} {{- $cache.Set "festivals" $byFestival -}} {{- $cache.Set "venues" $byVenue -}} {{- $cache.Set "albums" $byAlbum -}} {{- return $cache -}}
