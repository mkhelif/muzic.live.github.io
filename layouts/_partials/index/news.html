{{- /* Cached index of news, grouped by referenced entity. Returns a scratch with: - "artists" : artist id -> slice of news pages - "festivals" : festival id -> slice of news pages - "venues" : venue id -> slice of news pages - "albums" : album id -> slice of news pages Slices are sorted by date, most recent first. */ -}} {{- $cache := newScratch -}} {{- $cache.Set "artists" dict -}} {{- $cache.Set "festivals" dict -}} {{- $cache.Set "venues" dict -}} {{- $cache.Set "albums" dict -}} {{- range $news := (where site.RegularPages "Section" "news").ByDate.Reverse -}} {{- range $news.Params.artists -}} {{- $existing := index ($cache.Get "artists") . | default slice -}} {{- $cache.SetInMap "artists" . ($existing | append $news) -}} {{- end -}} {{- with $news.Params.festival -}} {{- $existing := index ($cache.Get "festivals") . | default slice -}} {{- $cache.SetInMap "festivals" . ($existing | append $news) -}} {{- end -}} {{- with $news.Params.venue -}} {{- $existing := index ($cache.Get "venues") . | default slice -}} {{- $cache.SetInMap "venues" . ($existing | append $news) -}} {{- end -}} {{- with $news.Params.album -}} {{- $existing := index ($cache.Get "albums") . | default slice -}} {{- $cache.SetInMap "albums" . ($existing | append $news) -}} {{- end -}} {{- end -}} {{- return $cache -}}
